
class DonnyBoyParser extends FirmwareParser {
  DonnyBoyParser._();
  static final DonnyBoyParser instance = DonnyBoyParser._();

  @override
  String get directoryUrl =>
      'https://api.github.com/repos/luciferbelzebob3-afk/DONNYBOYCZ/releases';

  @override
  Future<FirmwareDirectory> fetch() async {
    final raw = await fetchJson(Uri.parse(directoryUrl));
    final result = parseReleases(raw);
    seedCache(result);
    return result;
  }

  static FirmwareDirectory parseReleases(dynamic raw) {
    if (raw is! List) {
      throw FirmwareDirectoryUnreadable.one('GitHub releases is not a list');
    }
    final stable = <FirmwareVersion>[];
    final development = <FirmwareVersion>[];
    for (final release in raw) {
      if (release is! Map<String, dynamic> || release['draft'] == true) continue;
      final tag = release['tag_name'];
      final assets = release['assets'];
      if (tag is! String || assets is! List) continue;
      final files = <FirmwareFile>[];
      for (final asset in assets) {
        if (asset is! Map<String, dynamic>) continue;
        final name = asset['name'];
        final url = asset['browser_download_url'];
        final digest = asset['digest'];
        if (name is! String || url is! String || digest is! String) continue;
        if (!RegExp(r'^flipper-z-f7-update-.+\.tgz$').hasMatch(name)) continue;
        if (!RegExp(r'^sha256:[0-9a-fA-F]{64}$').hasMatch(digest)) continue;
        final uri = Uri.tryParse(url);
        if (uri == null || uri.scheme != 'https' || uri.host != 'github.com' ||
            !uri.path.startsWith('/luciferbelzebob3-afk/DONNYBOYCZ/releases/download/')) continue;
        files.add(FirmwareFile(url: url, target: 'f7', type: 'update_tgz',
          sha256: digest.substring(7).toLowerCase()));
      }
      if (files.isEmpty) continue;
      final version = FirmwareVersion(
        version: tag, changelog: release['body'] is String ? release['body'] as String : '',
        files: files,
      );
      (release['prerelease'] == true ? development : stable).add(version);
    }
    if (stable.isEmpty && development.isEmpty) {
      throw FirmwareDirectoryUnreadable.one('No verified DONNY BOY update packages');
    }
    return FirmwareDirectory(channels: [
      FirmwareDirectoryChannel(id: 'release', title: 'Stabilní',
        description: 'Oficiální vydání DONNY BOY CZ na GitHubu', versions: stable),
      FirmwareDirectoryChannel(id: 'development', title: 'Vývojové',
        description: 'Testovací vydání DONNY BOY CZ', versions: development),
    ]);
  }
}
